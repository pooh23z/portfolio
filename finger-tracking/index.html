Air Canvas & Dual-Hand Finger Tracking

ระบบวาดภาพในอากาศไร้สัมผัส ตรวจจับและแยกแยะมือซ้าย-ขวาแบบเรียลไทม์ ด้วย Python, OpenCV และ MediaPipe

✌️

Dual-Hand Tracking

รองรับการตรวจจับและระบุพิกัด 21 จุดสำคัญ (Landmarks) บนมือ 2 ข้างพร้อมกันด้วยความแม่นยำสูง

🎨

Air Canvas Drawing

วาดรูปกลางอากาศด้วยปลายนิ้วชี้ เลือกเปลี่ยนสีพู่กัน หรือเลือกยางลบโดยใช้ภาษามือ (Gesture Commands)

⚡

Real-time Performance

ประมวลผลวิดีโอจากกล้อง Webcam แบบ Frame-by-Frame ทำงานลื่นไหล ไม่หน่วง

🎨 Interactive Air Canvas Demo (ทดลองวาด)

คลิกลากลากเมาส์บนผืนผ้าใบเพื่อทดลองวาดภาพ

🐍 Python Source Code (MediaPipe + OpenCV)

import cv2
import mediapipe as mp
import numpy as np
import math

# 1. ตั้งค่า MediaPipe
mp_hands = mp.solutions.hands
hands = mp_hands.Hands(
    static_image_mode=False,
    max_num_hands=2, # รองรับ 2 มือพร้อมกัน
    min_detection_confidence=0.7,
    min_tracking_confidence=0.7
)
mp_draw = mp.solutions.drawing_utils

# 2. เปิดกล้อง Webcam
cap = cv2.VideoCapture(0)

px = {"Left": 0, "Right": 0}
py = {"Left": 0, "Right": 0}
canvas = None

color_red = (0, 0, 255)
color_green = (0, 255, 0)
color_blue = (255, 0, 0)
color_yellow = (0, 255, 255)
color_eraser = (0, 0, 0)

current_color = {"Left": color_red, "Right": color_green}
brush_thickness = 10
eraser_thickness = 50

while True:
    success, img = cap.read()
    if not success:
        break

    img = cv2.flip(img, 1)
    h, w, c = img.shape
    
    if canvas is None:
        canvas = np.zeros_like(img)

    # วาดแถบเมนูสีด้านบน
    menu_w = w // 5
    cv2.rectangle(img, (0, 0), (menu_w, 80), color_red, cv2.FILLED)
    cv2.rectangle(img, (menu_w, 0), (menu_w*2, 80), color_green, cv2.FILLED)
    cv2.rectangle(img, (menu_w*2, 0), (menu_w*3, 80), color_blue, cv2.FILLED)
    cv2.rectangle(img, (menu_w*3, 0), (menu_w*4, 80), color_yellow, cv2.FILLED)
    cv2.rectangle(img, (menu_w*4, 0), (w, 80), (50, 50, 50), cv2.FILLED)
    cv2.putText(img, "ERASER", (menu_w*4 + 10, 50), cv2.FONT_HERSHEY_SIMPLEX, 0.8, (255, 255, 255), 2)

    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    results = hands.process(img_rgb)

    if results.multi_hand_landmarks:
        for hand_idx, hand_landmarks in enumerate(results.multi_hand_landmarks):
            mp_draw.draw_landmarks(img, hand_landmarks, mp_hands.HAND_CONNECTIONS)
            hand_label = results.multi_handedness[hand_idx].classification[0].label
            
            lm_list = []
            for id, lm in enumerate(hand_landmarks.landmark):
                cx, cy = int(lm.x * w), int(lm.y * h)
                lm_list.append((cx, cy))
                
            if len(lm_list) != 0:
                x1, y1 = lm_list[8]  # ปลายนิ้วชี้
                x2, y2 = lm_list[12] # ปลายนิ้วกลาง
                
                fingers_up = [
                    lm_list[8][1] < lm_list[6][1],
                    lm_list[12][1] < lm_list[10][1]
                ]
                
                # โหมดเลือกสี (ชู 2 นิ้ว)
                if fingers_up[0] and fingers_up[1]:
                    px[hand_label], py[hand_label] = 0, 0
                    if y1 < 80:
                        if 0 < x1 < menu_w: current_color[hand_label] = color_red
                        elif menu_w < x1 < menu_w*2: current_color[hand_label] = color_green
                        elif menu_w*2 < x1 < menu_w*3: current_color[hand_label] = color_blue
                        elif menu_w*3 < x1 < menu_w*4: current_color[hand_label] = color_yellow
                        elif menu_w*4 < x1 < w: current_color[hand_label] = color_eraser
                        
                # โหมดวาด (ชูนิ้วชี้เดียว)
                elif fingers_up[0] and not fingers_up[1]:
                    if px[hand_label] == 0 and py[hand_label] == 0:
                        px[hand_label], py[hand_label] = x1, y1
                        
                    thick = eraser_thickness if current_color[hand_label] == color_eraser else brush_thickness
                    cv2.line(canvas, (px[hand_label], py[hand_label]), (x1, y1), current_color[hand_label], thick)
                    px[hand_label], py[hand_label] = x1, y1

    # ผสมภาพกล้องกับผืนผ้าใบ
    img_gray = cv2.cvtColor(canvas, cv2.COLOR_BGR2GRAY)
    _, img_inv = cv2.threshold(img_gray, 50, 255, cv2.THRESH_BINARY_INV)
    img_inv = cv2.cvtColor(img_inv, cv2.COLOR_GRAY2BGR)
    img = cv2.bitwise_and(img, img_inv)
    img = cv2.bitwise_or(img, canvas)

    cv2.imshow("Air Canvas Dual-Hand", img)
    if cv2.waitKey(1) & 0xFF == 27: # กด Esc เพื่อออก
        break

cap.release()
cv2.destroyAllWindows()
